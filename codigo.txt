def normalize_uuid(uuid_str: str) -> str:
    cleaned = uuid_str.lower().strip()
    if cleaned.startswith("0000") and cleaned.endswith("-0000-1000-8000-00805f9b34fb"):
        return cleaned[4:8]
    return cleaned

    # MARK: - CoreBluetooth Manager Helper (macOS Native)

    async def _get_central_manager(self) -> Optional[Any]:
        """Obtiene o inicializa el CentralManagerDelegate de CoreBluetooth en macOS."""
        if not HAS_MACOS_COREBLUETOOTH:
            return None
        if self._manager is None:
            try:
                self._manager = CentralManagerDelegate()
                await self._manager.wait_until_ready()
            except Exception as e:
                logger.warning(f"No se pudo inicializar CentralManagerDelegate: {e}")
                return None
        else:
            try:
                await self._manager.wait_until_ready()
            except Exception:
                pass
        return self._manager

    # MARK: - WebSocket Broadcasting

    async def register_websocket(self, ws: Any):
        self.websocket_clients.add(ws)

    async def unregister_websocket(self, ws: Any):
        self.websocket_clients.discard(ws)

    async def broadcast_event(self, event_type: str, data: Any):
        if not self.websocket_clients:
            return
        payload = json.dumps({"type": event_type, "data": data, "timestamp": datetime.now().isoformat()})
        disconnected = set()
        for ws in self.websocket_clients:
            try:
                await ws.send_text(payload)
            except Exception:
                disconnected.add(ws)
        self.websocket_clients.difference_update(disconnected)

    # MARK: - Recuperar Reloj Ya Conectado al Mac (macOS Native)

    async def retrieve_connected_peripherals(self) -> List[BLEDevice]:
        """
        Consulta directamente a CoreBluetooth por periféricos que YA están conectados
        o enlazados a macOS Monterey, permitiendo conectar inmediatamente a la Galaxy Fit3 (78DC)
        incluso si ha dejado de emitir anuncios BLE.
        """
        if not HAS_MACOS_COREBLUETOOTH:
            return []

        manager = await self._get_central_manager()
        if not manager or not hasattr(manager, "central_manager"):
            return []

        retrieved_list: List[BLEDevice] = []
        seen_addresses: Set[str] = set()

        # 1. retrieveConnectedPeripheralsWithServices_
        try:
            cbuuids = [CBUUID.UUIDWithString_(s) for s in CONNECTED_QUERY_SERVICES]
            connected_peripherals = manager.central_manager.retrieveConnectedPeripheralsWithServices_(cbuuids)
            if connected_peripherals:
                for p in connected_peripherals:
                    addr = str(p.identifier().UUIDString())
                    if addr in seen_addresses:
                        continue
                    seen_addresses.add(addr)
                    raw_name = str(p.name()) if p.name() else "Galaxy Fit3 (78DC)"
                    is_connected = (p.state() == CBPeripheralStateConnected)
                    
                    # Construir objeto BLEDevice real con details = (CBPeripheral, CentralManagerDelegate)
                    device_obj = BLEDevice(
                        address=addr,
                        name=raw_name,
                        details=(p, manager)
                    )
                    self.discovered_devices[addr] = device_obj
                    retrieved_list.append(device_obj)

                    upper_name = raw_name.upper()
                    is_fit3 = (
                        "FIT3" in upper_name
                        or "78DC" in upper_name
                        or "SM-R390" in upper_name
                        or "GALAXY" in upper_name
                        or "SAMSUNG" in upper_name
                    )

                    self.scanned_devices[addr] = {
                        "address": addr,
                        "name": raw_name,
                        "rssi": -45 if is_connected else -58,
                        "is_fit3": is_fit3,
                        "is_connected_system": is_connected,
                        "system_status": "Ya conectado a macOS" if is_connected else "Enlazado en el Mac",
                        "services": ["180D", "180F", "FD5A"],
                        "last_seen": datetime.now().isoformat()
                    }
                    logger.info(f"✅ Periférico recuperado de CoreBluetooth: {raw_name} ({addr}) - Estado: {is_connected}")
        except Exception as e:
            logger.warning(f"Error en retrieveConnectedPeripheralsWithServices_: {e}")

        # 2. retrievePeripheralsWithIdentifiers_ usando UUID guardado previamente
        saved_uuid = database.get_setting("last_fit3_uuid")
        if saved_uuid and saved_uuid not in seen_addresses:
            try:
                nsuuid = NSUUID.alloc().initWithUUIDString_(saved_uuid)
                if nsuuid:
                    known_peripherals = manager.central_manager.retrievePeripheralsWithIdentifiers_([nsuuid])
                    if known_peripherals:
                        for p in known_peripherals:
                            addr = str(p.identifier().UUIDString())
                            if addr in seen_addresses:
                                continue
                            seen_addresses.add(addr)
                            saved_name = database.get_setting("last_fit3_name", "Galaxy Fit3 (78DC)")
                            raw_name = str(p.name()) if p.name() else saved_name
                            is_connected = (p.state() == CBPeripheralStateConnected)

                            device_obj = BLEDevice(
                                address=addr,
                                name=raw_name,
                                details=(p, manager)
                            )
                            self.discovered_devices[addr] = device_obj
                            retrieved_list.append(device_obj)

                            self.scanned_devices[addr] = {
                                "address": addr,
                                "name": raw_name,
                                "rssi": -45 if is_connected else -60,
                                "is_fit3": True,
                                "is_connected_system": is_connected,
                                "system_status": "Ya conectado a macOS" if is_connected else "Enlazado previamente",
                                "services": ["180D", "180F", "FD5A"],
                                "last_seen": datetime.now().isoformat()
                            }
                            logger.info(f"✅ Periférico recuperado por NSUUID guardado: {raw_name} ({addr})")
            except Exception as e:
                logger.debug(f"Error en retrievePeripheralsWithIdentifiers_: {e}")

        return retrieved_list

    # MARK: - BLE Scanning

    async def scan_devices(self, timeout: float = 5.0) -> List[Dict[str, Any]]:
        if self.is_mock_mode:
            return self.get_scanned_devices_list()

        self.connection_state = "scanning"
        self.last_status_message = "Escaneando dispositivos BLE cercanos..."
        await self.broadcast_event("status_update", self.get_status())

        # Limpiar lista anterior para no acumular datos obsoletos
        self.scanned_devices.clear()
        self.discovered_devices.clear()

        # PASO A: Recuperar periféricos que ya están conectados o guardados en macOS
        retrieved_mac = await self.retrieve_connected_peripherals()
        if retrieved_mac:
            self.last_status_message = f"Se detectó {len(retrieved_mac)} reloj/dispositivo ya enlazado al Mac. Escaneando anuncios..."
            await self.broadcast_event("status_update", self.get_status())

        # PASO B: Escaneo estándar de anuncios activos con BleakScanner
        try:
            discovered = await BleakScanner.discover(timeout=timeout, return_adv=True)
            for address, (device, adv_data) in discovered.items():
                # Guardar el objeto BLEDevice real en cache
                self.discovered_devices[address] = device

                name = device.name or adv_data.local_name or "Desconocido"
                upper_name = name.upper()
                is_fit3 = (
                    "FIT3" in upper_name
                    or "78DC" in upper_name
                    or "SM-R390" in upper_name
                    or "GALAXY" in upper_name
                    or "SAMSUNG" in upper_name
                )
                service_uuids = [normalize_uuid(u) for u in adv_data.service_uuids]

                # Si ya existía por retrieveConnectedPeripherals, preservar su marca de conexión
                prev = self.scanned_devices.get(address)
                is_conn_sys = prev.get("is_connected_system", False) if prev else False

                self.scanned_devices[address] = {
                    "address": address,
                    "name": name,
                    "rssi": adv_data.rssi if adv_data.rssi is not None else -70,
                    "is_fit3": is_fit3 or (prev.get("is_fit3", False) if prev else False),
                    "is_connected_system": is_conn_sys,
                    "system_status": "Ya conectado a macOS" if is_conn_sys else "Anunciando vía BLE",
                    "services": service_uuids or (prev.get("services", []) if prev else []),
                    "last_seen": datetime.now().isoformat()
                }

            self.connection_state = "connected" if (self.client and self.client.is_connected) else "disconnected"
            self.last_status_message = f"Escaneo completado. {len(self.scanned_devices)} dispositivos encontrados."
        except Exception as e:
            logger.error(f"Aviso durante escaneo de anuncios: {e}")
            self.connection_state = "connected" if (self.client and self.client.is_connected) else "disconnected"
            self.last_status_message = f"Escaneo finalizado ({len(self.scanned_devices)} dispositivos listos)."

        await self.broadcast_event("status_update", self.get_status())
        return self.get_scanned_devices_list()

    def get_scanned_devices_list(self) -> List[Dict[str, Any]]:
        devices: List[Dict[str, Any]] = []
        for d in self.scanned_devices.values():
            addr = d.get("address", "")
            # REGLA 3: Si el modo simulación está apagado, jamás mostrar dispositivos con direcciones mock
            if not self.is_mock_mode:
                if "MOCK" in addr or addr.startswith("AA:BB") or addr == "F9:A2:78:DC:39:00":
                    continue
            devices.append(d)

        # Destacar al inicio:
        # 1. Periféricos ya conectados a macOS
        # 2. Periféricos Galaxy Fit3
        # 3. Mayor RSSI
        devices.sort(key=lambda d: (
            not d.get("is_connected_system", False),
            not d.get("is_fit3", False),
            -d.get("rssi", -100)
        ))
        return devices

    # MARK: - Connection Management (Pass Real BLEDevice)

    async def connect(self, address: str) -> bool:
        async with self._connect_lock:
            if self.is_mock_mode:
                await self.toggle_mock_mode(False)

            # Si ya estamos conectados a esta misma dirección, retornar éxito inmediato
            if self.client and self.client.is_connected and self.connected_device_address == address:
                logger.info(f"Ya conectado previamente a {address}.")
                return True

            if self._reconnect_task:
                self._reconnect_task.cancel()
                self._reconnect_task = None

            if self.client and self.client.is_connected:
                await self.disconnect()

            # REGLA 1: Obtener el objeto BLEDevice real cacheado
            ble_device = self.discovered_devices.get(address)

            # Si no estaba en cache (por ejemplo, tras reinicio), intentar recuperarlo nativamente vía CoreBluetooth
            if ble_device is None:
                logger.info(f"Buscando BLEDevice para {address} mediante CoreBluetooth...")
                await self.retrieve_connected_peripherals()
                ble_device = self.discovered_devices.get(address)

            if ble_device is None:
                err = f"No se encontró el objeto BLEDevice para '{address}'. Ejecuta un escaneo primero."
                logger.error(err)
                self.last_status_message = err
                await self.broadcast_event("status_update", self.get_status())
                raise Exception(err)

            self.connection_state = "connecting"
            device_name = ble_device.name or self.scanned_devices.get(address, {}).get("name", "Galaxy Fit3 (78DC)")
            self.last_status_message = f"Conectando a {device_name} ({address})..."
            await self.broadcast_event("status_update", self.get_status())

            def on_disconnect(client: Any = None):
                addr = getattr(client, "address", address) if client else address
                logger.warning(f"Periférico {addr} desconectado.")
                if self.connection_state == "connected":
                    self.connection_state = "disconnected"
                    self.last_status_message = "Conexión perdida con el reloj. Intentando reconexión..."
                    asyncio.create_task(self.broadcast_event("status_update", self.get_status()))
                    if not self._reconnect_task or self._reconnect_task.done():
                        self._reconnect_task = asyncio.create_task(self._auto_reconnect_loop(address))

            try:
                logger.info(f"Instanciando BleakClient con BLEDevice real: {ble_device} (details={ble_device.details})")
                
                # CRÍTICO: Referencia fuerte al CBPeripheral en la clase para evitar recolección por ARC/PyObjC
                if ble_device.details and isinstance(ble_device.details, tuple):
                    self._active_peripheral = ble_device.details[0]

                self.client = BleakClient(
                    ble_device,
                    disconnected_callback=on_disconnect,
                    timeout=20.0
                )
                
                # CRÍTICO: En Bleak 1.0+, connect() retorna None (no booleano) si tiene éxito y lanza excepción si falla.
                await self.client.connect()
                
                # Validar estado mediante is_connected
                if self.client.is_connected:
                    self.connected_device_address = address
                    self.connected_device_name = device_name
                    self.connection_state = "connected"
                    self.last_status_message = f"Conectado a {self.connected_device_name}. Enumerando servicios GATT..."
                    logger.info(f"✅ Conexión exitosa con {self.connected_device_name} ({address})")
                    
                    # Guardar UUID en base de datos para recuperación automática instantánea
                    database.save_setting("last_fit3_uuid", address)
                    database.save_setting("last_fit3_name", device_name)

                    await self.broadcast_event("status_update", self.get_status())

                    # Enumerar servicios y características
                    await self._discover_gatt_services()

                    # Suscripción automática a Heart Rate y Batería si están disponibles
                    await self._auto_subscribe_essential_services()

                    return True
                else:
                    raise Exception("El cliente BLE no confirmó el estado conectado (is_connected=False).")
            except Exception as e:
                tb = traceback.format_exc()
                logger.error(f"❌ Error detallado al conectar con {address}:\n{tb}")
                self.connection_state = "disconnected"
                self._active_peripheral = None
                if self.client:
                    try:
                        await self.client.disconnect()
                    except Exception:
                        pass
                    self.client = None
                
                raw_err = str(e) or type(e).__name__
                hint = ""
                lower_err = raw_err.lower()
                if "denied" in lower_err or "unauthorized" in lower_err:
                    hint = " [Permiso denegado: Ve a Ajustes del Sistema > Privacidad y Seguridad > Bluetooth y habilita Terminal / Python]"
                elif "not found" in lower_err:
                    hint = " [Dispositivo no encontrado en BLE: Asegúrate de que el reloj esté encendido o enlazado al Mac]"

                self.last_status_message = f"Error al conectar: {raw_err}{hint}"
                await self.broadcast_event("status_update", self.get_status())
                raise Exception(self.last_status_message)

    async def disconnect(self):
        if self._reconnect_task:
            self._reconnect_task.cancel()
            self._reconnect_task = None

        if self.client:
            try:
                if self.client.is_connected:
                    await self.client.disconnect()
            except Exception as e:
                logger.debug(f"Aviso al desconectar: {e}")
            finally:
                self.client = None

        self._active_peripheral = None
        self.connected_device_address = None
        self.connection_state = "disconnected"
        self.active_subscriptions.clear()
        self.last_status_message = "Reloj desconectado."
        await self.broadcast_event("status_update", self.get_status())

    async def _auto_reconnect_loop(self, address: str):
        """Intenta reconectar automáticamente si el reloj se aleja y vuelve."""
        retry_interval = 4.0
        while self.connection_state == "disconnected" and not self.is_mock_mode:
            logger.info(f"Reintentando conexión con {address}...")
            self.last_status_message = "Buscando Galaxy Fit 3 para reconexión automática..."
            await self.broadcast_event("status_update", self.get_status())
            try:
                # Primero recuperar de CoreBluetooth
                await self.retrieve_connected_peripherals()
                if address in self.discovered_devices:
                    ok = await self.connect(address)
                    if ok:
                        logger.info("Reconexión automática exitosa!")
                        return
            except Exception as e:
                logger.debug(f"Fallo temporal en reconexión: {e}")
            await asyncio.sleep(retry_interval)

    # MARK: - GATT Discovery & Subscriptions

    async def _discover_gatt_services(self):
        if not self.client or not self.client.is_connected:
            return

        self.discovered_services.clear()
        services = self.client.services

        for service in services:
            short_s_uuid = normalize_uuid(service.uuid)
            s_name = STANDARD_UUIDS.get(short_s_uuid, f"Servicio Propietario ({short_s_uuid.upper()})")
            is_proprietary = short_s_uuid not in STANDARD_UUIDS

            chars_info = []
            for char in service.characteristics:
                short_c_uuid = normalize_uuid(char.uuid)
                c_name = STANDARD_UUIDS.get(short_c_uuid, f"Característica {short_c_uuid.upper()}")
                props = char.properties

                can_read = "read" in props
                can_write = ("write" in props) or ("write-without-response" in props)
                can_notify = ("notify" in props) or ("indicate" in props)

                # Si es una característica estándar segura de información, leer su valor inicial
                if can_read and short_c_uuid in ["2a00", "2a19", "2a24", "2a26", "2a29", "2a38"]:
                    try:
                        val = await asyncio.wait_for(self.client.read_gatt_char(char.uuid), timeout=1.5)
                        last_hex = val.hex().upper()
                        self._process_characteristic_data(service.uuid, char.uuid, val, direction="RX", note="Lectura inicial")
                    except Exception as e:
                        logger.debug(f"Lectura inicial omitida para {char.uuid}: {e}")

                chars_info.append({
                    "service_uuid": service.uuid,
                    "uuid": char.uuid,
                    "short_uuid": short_c_uuid.upper(),
                    "name": c_name,
                    "properties": props,
                    "can_read": can_read,
                    "can_write": can_write,
                    "can_notify": can_notify,
                    "is_notifying": char.uuid in self.active_subscriptions,
                    "last_hex": last_hex
                })

            self.discovered_services.append({
                "uuid": service.uuid,
                "short_uuid": short_s_uuid.upper(),
                "name": s_name,
                "is_proprietary": is_proprietary,
                "characteristics": chars_info
            })

        self.last_status_message = f"GATT listo ({len(self.discovered_services)} servicios enumerados)."
        await self.broadcast_event("services_discovered", self.discovered_services)

    async def _auto_subscribe_essential_services(self):
        """
        1. Lee directamente el nivel de batería real en 0x2A19 (GATT estándar 0x180F).
        2. Suscribe notificaciones a Heart Rate (0x2A37) y Batería (0x2A19).
        3. Suscribe automáticamente a todas las características notificables de los servicios propietarios de Samsung (UUIDs FEE..., FD..., etc.).
        """
        if not self.client or not self.client.is_connected:
            return

        # PASO 1: Lectura directa de Batería real (0x2A19)
        for s in self.discovered_services:
            for c in s["characteristics"]:
                short_c = c["short_uuid"].lower()
                full_c = c["uuid"].lower()
                if short_c == "2a19" or "00002a19" in full_c:
                    try:
                        logger.info(f"🔋 Leyendo porcentaje de batería real en {c['uuid']}...")
                        val = await asyncio.wait_for(self.client.read_gatt_char(c["uuid"]), timeout=3.0)
                        if val and len(val) >= 1:
                            real_bat = int(val[0])
                            if 0 <= real_bat <= 100:
                                self.battery_level = real_bat
                                self.has_real_battery = True
                                database.save_battery(self.battery_level)
                                c["last_hex"] = val.hex().upper()
                                logger.info(f"✅ Batería real confirmada del reloj: {self.battery_level}%")
                                await self.broadcast_event("telemetry_update", self.get_telemetry())
                    except Exception as e:
                        logger.warning(f"Aviso al leer batería directa 0x2A19: {e}")

        # PASO 2: Suscribir a Heart Rate, Batería y Servicios Propietarios de Samsung
        for s in self.discovered_services:
            short_s = s["short_uuid"].lower()
            is_samsung_service = (
                s.get("is_proprietary", False)
                or short_s.startswith("fe")
                or short_s.startswith("fd")
                or "0000fee" in s["uuid"].lower()
                or "0000fd" in s["uuid"].lower()
            )

            for c in s["characteristics"]:
                short_c = c["short_uuid"].lower()
                full_c = c["uuid"].lower()
                can_notify = c.get("can_notify", False)

                if not can_notify:
                    continue

                should_subscribe = False
                # Estándares esenciales
                if short_c in ["2a37", "2a19", "2a53"]:
                    should_subscribe = True
                # Propietarios Samsung (FEE..., FD...)
                elif is_samsung_service or short_c.startswith("fe") or short_c.startswith("fd") or "fee" in full_c or "fd" in full_c:
                    should_subscribe = True

                if should_subscribe and c["uuid"] not in self.active_subscriptions:
                    logger.info(f"📡 Suscribiendo automáticamente a característica: {c['name']} ({c['uuid']})")
                    await self.start_notify(c["uuid"])
                    await asyncio.sleep(0.08)

        await self.broadcast_event("telemetry_update", self.get_telemetry())
        await self.broadcast_event("status_update", self.get_status())

    async def start_notify(self, char_uuid: str) -> bool:
        if not self.client or not self.client.is_connected:
            return False

        def notification_handler(sender: BleakGATTCharacteristic, data: bytearray):
            raw = bytes(data)
            self._process_characteristic_data(
                service_uuid=sender.service_uuid if hasattr(sender, "service_uuid") else "Unknown",
                char_uuid=sender.uuid,
                data=raw,
                direction="RX",
                note="Notificación en vivo"
            )

        try:
            await self.client.start_notify(char_uuid, notification_handler)
            self.active_subscriptions.add(char_uuid)
            self._update_char_notify_state(char_uuid, True)
            self.last_status_message = f"Suscripción activa para {char_uuid}."
            await self.broadcast_event("status_update", self.get_status())
            return True
        except Exception as e:
            logger.error(f"Error al suscribir notificaciones en {char_uuid}: {e}")
            return False

    async def stop_notify(self, char_uuid: str) -> bool:
        if not self.client or not self.client.is_connected:
            return False
        try:
            await self.client.stop_notify(char_uuid)
            self.active_subscriptions.discard(char_uuid)
            self._update_char_notify_state(char_uuid, False)
            self.last_status_message = f"Suscripción detenida en {char_uuid}."
            await self.broadcast_event("status_update", self.get_status())
            return True
        except Exception as e:
            logger.error(f"Error al detener notificación {char_uuid}: {e}")
            return False

    async def subscribe_all_notifiable(self) -> int:
        subscribed_count = 0
        for s in self.discovered_services:
            for c in s["characteristics"]:
                if c["can_notify"] and c["uuid"] not in self.active_subscriptions:
                    ok = await self.start_notify(c["uuid"])
                    if ok:
                        subscribed_count += 1
                        await asyncio.sleep(0.1)
        self.last_status_message = f"Se activó notificación en {subscribed_count} características."
        await self.broadcast_event("status_update", self.get_status())
        return subscribed_count

    async def read_characteristic(self, char_uuid: str) -> Optional[str]:
        if not self.client or not self.client.is_connected:
            return None
        try:
            val = await self.client.read_gatt_char(char_uuid)
            raw = bytes(val)
            self._process_characteristic_data(service_uuid="ReadCmd", char_uuid=char_uuid, data=raw, direction="RX", note="Lectura manual")
            return raw.hex().upper()
        except Exception as e:
            logger.error(f"Error al leer característica {char_uuid}: {e}")
            return None

    async def write_characteristic(self, char_uuid: str, hex_string: str) -> bool:
        cleaned_hex = hex_string.replace("0x", "").replace(" ", "").replace("-", "")
        try:
            data = bytes.fromhex(cleaned_hex)
        except ValueError:
            self.last_status_message = "Formato Hexadecimal inválido."
            return False

        if self.is_mock_mode:
            self._process_characteristic_data("MockService", char_uuid, data, direction="TX", note="Envío manual simulado")
            self.last_status_message = f"TX simulado ({len(data)} bytes) hacia {char_uuid}."
            await self.broadcast_event("status_update", self.get_status())
            return True

        if not self.client or not self.client.is_connected:
            self.last_status_message = "Reloj no conectado para escribir."
            return False

        try:
            await self.client.write_gatt_char(char_uuid, data)
            self._process_characteristic_data("CustomCmd", char_uuid, data, direction="TX", note="Envío manual Hex")
            self.last_status_message = f"Enviados {len(data)} bytes a {char_uuid}."
            await self.broadcast_event("status_update", self.get_status())
            return True
        except Exception as e:
            logger.error(f"Error al escribir en {char_uuid}: {e}")
            self.last_status_message = f"Error al escribir: {str(e)}"
            return False

    # MARK: - Packet Parsing & Reverse Engineering

    def _analyze_step_candidates(self, char_uuid: str, byte_list: List[int]) -> List[Dict[str, Any]]:
        """
        Analiza secuencias de 2 y 4 bytes Little-Endian (uint16 / uint32) en cada offset.
        Compara con el payload anterior de esta misma característica para calcular deltas.
        Identifica secuencias que aumentan progresivamente (+1 a +150) o valores en rango de pasos diarios.
        """
        prev_bytes = self._last_char_payloads.get(char_uuid)
        candidates = []
        n = len(byte_list)

        # 1. uint16 Little-Endian en offsets 0..n-2
        for offset in range(0, n - 1):
            val16 = byte_list[offset] | (byte_list[offset + 1] << 8)
            delta = None
            is_inc = False
            if prev_bytes and len(prev_bytes) >= offset + 2:
                prev_val16 = prev_bytes[offset] | (prev_bytes[offset + 1] << 8)
                delta = val16 - prev_val16
                if 1 <= delta <= 150:
                    is_inc = True

            if is_inc or (10 <= val16 <= 65000 and delta is not None and delta != 0):
                candidates.append({
                    "offset": offset,
                    "byte_len": 2,
                    "type": "uint16LE",
                    "value": val16,
                    "delta": delta,
                    "is_increment": is_inc
                })

        # 2. uint32 Little-Endian en offsets 0..n-4
        for offset in range(0, n - 3):
            val32 = (
                byte_list[offset]
                | (byte_list[offset + 1] << 8)
                | (byte_list[offset + 2] << 16)
                | (byte_list[offset + 3] << 24)
            )
            delta = None
            is_inc = False
            if prev_bytes and len(prev_bytes) >= offset + 4:
                prev_val32 = (
                    prev_bytes[offset]
                    | (prev_bytes[offset + 1] << 8)
                    | (prev_bytes[offset + 2] << 16)
                    | (prev_bytes[offset + 3] << 24)
                )
                delta = val32 - prev_val32
                if 1 <= delta <= 150:
                    is_inc = True

            if is_inc or (10 <= val32 <= 200000 and delta is not None and delta != 0):
                candidates.append({
                    "offset": offset,
                    "byte_len": 4,
                    "type": "uint32LE",
                    "value": val32,
                    "delta": delta,
                    "is_increment": is_inc
                })

        return candidates

    def _process_characteristic_data(self, service_uuid: str, char_uuid: str, data: bytes, direction: str = "RX", note: Optional[str] = None):
        if not data:
            return

        byte_list = list(data)
        hex_str = " ".join(f"{b:02X}" for b in byte_list)
        short_c = normalize_uuid(char_uuid).upper()

        # Análisis de secuencias Little-Endian y candidatos para Sniffer
        step_candidates = self._analyze_step_candidates(char_uuid, byte_list)
        decoded_summary = self._decode_little_endian(byte_list, step_candidates)

        # 1. Heart Rate GATT Estándar (0x2A37)
        if short_c == "2A37" or short_c == self.custom_hr_uuid.upper():
            flags = byte_list[0]
            is_16bit = (flags & 0x01) != 0
            contact = bool(flags & 0x02)
            bpm = 0
            offset = 1
            if is_16bit and len(byte_list) >= 3:
                bpm = byte_list[1] | (byte_list[2] << 8)
                offset = 3
            elif len(byte_list) >= 2:
                bpm = byte_list[1]
                offset = 2

            rr_ms = None
            if (flags & 0x10) and len(byte_list) >= offset + 2:
                raw_rr = byte_list[offset] | (byte_list[offset + 1] << 8)
                rr_ms = int((raw_rr / 1024.0) * 1000.0)

            if 30 <= bpm <= 240:
                self.current_hr = bpm
                self.has_real_hr = True
                database.save_heart_rate(bpm=bpm, contact=contact, rr_interval_ms=rr_ms, source="Galaxy Fit3 BLE (Confirmado)")
                note = f"Pulso real: {bpm} BPM"

        # 2. Battery Service GATT (0x2A19)
        elif short_c == "2A19" or "00002a19" in char_uuid.lower():
            level = byte_list[0]
            if 0 <= level <= 100:
                self.battery_level = level
                self.has_real_battery = True
                database.save_battery(self.battery_level)
                note = f"Batería real: {self.battery_level}%"

        # 3. Device Info (0x2A24 / 0x2A26 / 0x2A29)
        elif short_c == "2A24":
            self.device_model = data.decode(errors="ignore").strip()
            note = f"Modelo: {self.device_model}"
        elif short_c == "2A26":
            self.device_firmware = data.decode(errors="ignore").strip()
            note = f"Firmware: {self.device_firmware}"
        elif short_c == "2A29":
            self.device_manufacturer = data.decode(errors="ignore").strip()
            note = f"Fabricante: {self.device_manufacturer}"

        # 4. Mapeo Configurable de Pasos (UUID / Offset / Bytes seleccionados en la UI)
        is_mapped_steps_char = (
            short_c == self.custom_steps_uuid.upper()
            or char_uuid.lower() == self.custom_steps_uuid.lower()
            or (self.custom_steps_uuid.upper() in short_c)
        )

        if is_mapped_steps_char:
            offset = self.custom_steps_offset
            byte_len = self.custom_steps_bytes # 2 o 4
            steps = None
            if byte_len == 2 and len(byte_list) >= offset + 2:
                steps = byte_list[offset] | (byte_list[offset + 1] << 8)
            elif byte_len == 4 and len(byte_list) >= offset + 4:
                steps = (
                    byte_list[offset]
                    | (byte_list[offset + 1] << 8)
                    | (byte_list[offset + 2] << 16)
                    | (byte_list[offset + 3] << 24)
                )

            if steps is not None and 0 <= steps <= 250000:
                self.today_steps = steps
                self.has_real_steps = True
                self.today_calories = round(steps * 0.042, 1)
                self.today_distance = round(steps * 0.74, 1)
                self.today_active_minutes = max(self.today_active_minutes or 0, int(steps / 115))
                database.save_steps(
                    steps=self.today_steps,
                    calories=self.today_calories,
                    distance=self.today_distance,
                    active_minutes=self.today_active_minutes,
                    source="Galaxy Fit3 BLE (Confirmado)"
                )
                note = f"Pasos confirmados: {steps}"

        # Fallback heurístico para tramas con prefijo Samsung FE A1
        elif len(byte_list) >= 6 and byte_list[0] == 0xFE and byte_list[1] == 0xA1:
            steps = (
                byte_list[2]
                | (byte_list[3] << 8)
                | (byte_list[4] << 16)
                | (byte_list[5] << 24)
            )
            if 0 <= steps <= 250000:
                self.today_steps = steps
                self.has_real_steps = True
                self.today_calories = round(steps * 0.042, 1)
                self.today_distance = round(steps * 0.74, 1)
                self.today_active_minutes = max(self.today_active_minutes or 0, int(steps / 115))
                database.save_steps(
                    steps=self.today_steps,
                    calories=self.today_calories,
                    distance=self.today_distance,
                    active_minutes=self.today_active_minutes,
                    source="Galaxy Fit3 BLE (Confirmado)"
                )
                note = f"Pasos decodificados: {self.today_steps}"

        # 5. Mapeo Heurístico de Fases de Sueño (FE A2 ...)
        elif (short_c == self.custom_sleep_uuid.upper()) or (len(byte_list) >= 6 and byte_list[0] == 0xFE and byte_list[1] == 0xA2):
            offset = 2 if (byte_list[0] == 0xFE and byte_list[1] == 0xA2) else 0
            if len(byte_list) >= offset + 4:
                deep = byte_list[offset]
                light = byte_list[offset + 1] * 2
                rem = byte_list[offset + 2]
                awake = byte_list[offset + 3]
                now_date = datetime.now().strftime("%Y-%m-%d")
                self.has_real_sleep = True
                database.save_sleep_session(
                    start_time=f"{now_date}T00:15:00",
                    end_time=f"{now_date}T07:20:00",
                    deep=deep,
                    light=light,
                    rem=rem,
                    awake=awake,
                    score=min(98, 65 + int(deep / 4)),
                    source="Galaxy Fit3 BLE (Confirmado)"
                )
                note = f"Sueño RX: Prof {deep}m / Lig {light}m / REM {rem}m"

        # 6. SpO2 Heurístico (FE A3 ...)
        elif len(byte_list) >= 3 and byte_list[0] == 0xFE and byte_list[1] == 0xA3:
            spo2 = byte_list[2]
            if 85 <= spo2 <= 100:
                self.current_spo2 = spo2
                self.has_real_spo2 = True
                database.save_spo2(spo2, source="Galaxy Fit3 BLE (Confirmado)")
                note = f"SpO2 real: {spo2}%"

        # Actualizar cache de último payload para comparación de deltas
        self._last_char_payloads[char_uuid] = bytes(byte_list)

        # Guardar paquete en SQLite
        database.save_ble_packet(
            service_uuid=service_uuid,
            char_uuid=char_uuid,
            direction=direction,
            hex_string=hex_str,
            raw_bytes=byte_list,
            decoded_summary=decoded_summary,
            note=note
        )

        for s in self.discovered_services:
            for c in s["characteristics"]:
                if c["uuid"] == char_uuid:
                    c["last_hex"] = hex_str

        packet_payload = {
            "timestamp": datetime.now().strftime("%H:%M:%S.%f")[:-3],
            "service_uuid": service_uuid,
            "char_uuid": char_uuid,
            "short_uuid": short_c,
            "direction": direction,
            "hex_string": hex_str,
            "raw_bytes": byte_list,
            "utf8_preview": self._bytes_to_utf8(byte_list),
            "decoded_summary": decoded_summary,
            "step_candidates": step_candidates,
            "note": note
        }
        asyncio.create_task(self.broadcast_event("ble_packet", packet_payload))
        asyncio.create_task(self.broadcast_event("telemetry_update", self.get_telemetry()))

    def _decode_little_endian(self, bytes_arr: List[int], candidates: Optional[List[Dict[str, Any]]] = None) -> str:
        if not bytes_arr:
            return "Vacío"
        parts = [f"u8[0]={bytes_arr[0]}"]
        if len(bytes_arr) >= 2:
            u16_0 = bytes_arr[0] | (bytes_arr[1] << 8)
            parts.append(f"u16LE[0]={u16_0}")
        if len(bytes_arr) >= 4:
            u16_2 = bytes_arr[2] | (bytes_arr[3] << 8)
            u32_0 = bytes_arr[0] | (bytes_arr[1] << 8) | (bytes_arr[2] << 16) | (bytes_arr[3] << 24)
            parts.append(f"u16LE[2]={u16_2}")
            parts.append(f"u32LE[0]={u32_0}")

        if candidates:
            inc_items = [c for c in candidates if c.get("is_increment")]
            if inc_items:
                top = inc_items[0]
                parts.append(f"⚡DELTA: +{top['delta']} (off {top['offset']} {top['type']}={top['value']})")

        return " | ".join(parts)

    def _bytes_to_utf8(self, bytes_arr: List[int]) -> str:
        chars = [chr(b) if 32 <= b <= 126 else "." for b in bytes_arr]
        return "".join(chars)

def save_steps(steps: int, calories: Optional[float] = None, distance: Optional[float] = None, active_minutes: Optional[int] = None, source: str = "Galaxy Fit3 (78DC)"):
    if calories is None:
        calories = round(steps * 0.042, 1)
    if distance is None:
        distance = round(steps * 0.74, 1)
    if active_minutes is None:
        active_minutes = max(0, int(steps / 115))

    today_str = date.today().isoformat()
    now_str = datetime.now().isoformat()

    conn = get_connection()
    cursor = conn.cursor()
    cursor.execute("""
    INSERT INTO steps (day_date, steps, calories, distance_meters, active_minutes, timestamp, source)
    VALUES (?, ?, ?, ?, ?, ?, ?)
    ON CONFLICT(day_date) DO UPDATE SET
        steps = excluded.steps,
        calories = excluded.calories,
        distance_meters = excluded.distance_meters,
        active_minutes = excluded.active_minutes,
        timestamp = excluded.timestamp,
        source = excluded.source
    """, (today_str, steps, calories, distance, active_minutes, now_str, source))
    conn.commit()
    conn.close()

def get_today_summary() -> Dict[str, Any]:
    today_str = date.today().isoformat()
    conn = get_connection()
    cursor = conn.cursor()

    # Steps
    cursor.execute("SELECT * FROM steps WHERE day_date = ? AND source NOT LIKE '%Demo%'", (today_str,))
    step_row = cursor.fetchone()
    has_real_steps = step_row is not None
    steps_data = dict(step_row) if step_row else {
        "steps": 0,
        "calories": 0.0,
        "distance_meters": 0.0,
        "active_minutes": 0
    }

    # Latest Heart Rate
    cursor.execute("SELECT bpm, timestamp FROM heart_rate WHERE source NOT LIKE '%Demo%' ORDER BY id DESC LIMIT 1")
    hr_row = cursor.fetchone()
    has_real_hr = hr_row is not None
    latest_hr = hr_row["bpm"] if hr_row else None
    latest_hr_time = hr_row["timestamp"] if hr_row else None

    # Avg Heart Rate Today
    cursor.execute("SELECT AVG(bpm) as avg_bpm FROM heart_rate WHERE timestamp >= ? AND source NOT LIKE '%Demo%'", (today_str + "T00:00:00",))
    avg_hr_row = cursor.fetchone()
    avg_hr = int(round(avg_hr_row["avg_bpm"])) if (avg_hr_row and avg_hr_row["avg_bpm"]) else latest_hr

    # Latest Battery
    cursor.execute("SELECT level FROM battery ORDER BY id DESC LIMIT 1")
    bat_row = cursor.fetchone()
    has_real_battery = bat_row is not None
    battery_level = bat_row["level"] if bat_row else None

    # Latest SpO2
    cursor.execute("SELECT percentage FROM spo2 WHERE source NOT LIKE '%Demo%' ORDER BY id DESC LIMIT 1")
    spo2_row = cursor.fetchone()
    has_real_spo2 = spo2_row is not None
    latest_spo2 = spo2_row["percentage"] if spo2_row else None

    # Sleep
    cursor.execute("SELECT * FROM sleep_sessions WHERE day_date = ? AND source NOT LIKE '%Demo%'", (today_str,))
    sleep_row = cursor.fetchone()
    if not sleep_row:
        # Check most recent sleep
        cursor.execute("SELECT * FROM sleep_sessions WHERE source NOT LIKE '%Demo%' ORDER BY id DESC LIMIT 1")
        sleep_row = cursor.fetchone()

    has_real_sleep = sleep_row is not None
    if sleep_row:
        sleep_data = dict(sleep_row)
        total_sleep_minutes = (sleep_data.get("deep_minutes", 0) +
                              sleep_data.get("light_minutes", 0) +
                              sleep_data.get("rem_minutes", 0))
        sleep_dict = {
            "total_minutes": total_sleep_minutes,
            "deep_minutes": sleep_data.get("deep_minutes", 0),
            "light_minutes": sleep_data.get("light_minutes", 0),
            "rem_minutes": sleep_data.get("rem_minutes", 0),
            "awake_minutes": sleep_data.get("awake_minutes", 0),
            "sleep_score": sleep_data.get("sleep_score", 0),
            "start_time": sleep_data.get("start_time"),
            "end_time": sleep_data.get("end_time")
        }
    else:
        sleep_dict = {
            "total_minutes": 0,
            "deep_minutes": 0,
            "light_minutes": 0,
            "rem_minutes": 0,
            "awake_minutes": 0,
            "sleep_score": 0,
            "start_time": None,
            "end_time": None
        }

    # Recent HR points for Chart.js
    cursor.execute("SELECT bpm, timestamp FROM heart_rate WHERE source NOT LIKE '%Demo%' ORDER BY id DESC LIMIT 20")
    hr_samples = [dict(r) for r in cursor.fetchall()]
    hr_samples.reverse()

    conn.close()

    return {
        "has_real_steps": has_real_steps,
        "has_real_hr": has_real_hr,
        "has_real_battery": has_real_battery,
        "has_real_spo2": has_real_spo2,
        "has_real_sleep": has_real_sleep,
        "steps": steps_data.get("steps", 0),
        "calories": steps_data.get("calories", 0.0),
        "distance_meters": steps_data.get("distance_meters", 0.0),
        "active_minutes": steps_data.get("active_minutes", 0),
        "heart_rate_latest": latest_hr,
        "heart_rate_avg": avg_hr,
        "heart_rate_latest_time": latest_hr_time,

        "spo2": latest_spo2,

    }

def get_local_ip() -> str:
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        s.connect(('10.255.255.255', 1))
        ip = s.getsockname()[0]
    except Exception:
        ip = '127.0.0.1'
    finally:
        s.close()
    return ip

@app.websocket("/ws/ble-logs")
async def websocket_logs_endpoint(websocket: WebSocket):
    await websocket.accept()
    await ble_manager.register_websocket(websocket)

    # Enviar estado inicial y últimos paquetes
    try:
        init_payload = {
            "type": "init",
            "status": ble_manager.get_status(),
            "telemetry": ble_manager.get_telemetry(),
            "services": ble_manager.discovered_services,
            "recent_packets": database.get_recent_packets(limit=35)
        }
        await websocket.send_json(init_payload)

        while True:
            # Mantener conexión viva y procesar posibles comandos del frontend
            data = await websocket.receive_text()
            if data == "ping":
                await websocket.send_text("pong")
    except WebSocketDisconnect:
        await ble_manager.unregister_websocket(websocket)
    except Exception as e:
        logger.debug(f"WebSocket cerrado: {e}")
        await ble_manager.unregister_websocket(websocket)

    // MARK: - WebSockets para Telemetría y Paquetes Hex en Vivo
    function initWebSocket() {
      const loc = window.location;
      const wsProtocol = loc.protocol === 'https:' ? 'wss:' : 'ws:';
      const wsUrl = `${wsProtocol}//${loc.host}/ws/ble-logs`;

      ws = new WebSocket(wsUrl);

      ws.onopen = () => {
        console.log("WebSocket BLE conectado.");
      };

      ws.onmessage = (event) => {
        try {
          const msg = JSON.parse(event.data);
          if (msg.type === 'init') {
            updateStatusUI(msg.status);
            updateTelemetryUI(msg.telemetry);
            renderServicesTree(msg.services);
            if (msg.recent_packets) {
              msg.recent_packets.forEach(p => appendPacketToConsole(p, false));
            }
          } else if (msg.type === 'telemetry_update') {
            updateTelemetryUI(msg.data);
          } else if (msg.type === 'status_update') {
            updateStatusUI(msg.data);
          } else if (msg.type === 'ble_packet') {
            appendPacketToConsole(msg.data, true);
          } else if (msg.type === 'services_discovered') {
            renderServicesTree(msg.data);
          }
        } catch (e) {
          console.error("Error al procesar mensaje WebSocket:", e);
        }
      };

      ws.onclose = () => {
        console.warn("WebSocket cerrado. Reintentando en 3s...");
        setTimeout(initWebSocket, 3000);
      };
    }

        if (inBed > 0) {
          document.getElementById('barSleepDeep').style.width = `${(s.deep_minutes / inBed) * 100}%`;
          document.getElementById('barSleepLight').style.width = `${(s.light_minutes / inBed) * 100}%`;
          document.getElementById('barSleepRem').style.width = `${(s.rem_minutes / inBed) * 100}%`;
          document.getElementById('barSleepAwake').style.width = `${(s.awake_minutes / inBed) * 100}%`;
        }

        document.getElementById('txtSleepDeep').textContent = `${Math.floor((s.deep_minutes || 0) / 60)}h ${(s.deep_minutes || 0) % 60}m`;
        document.getElementById('txtSleepLight').textContent = `${Math.floor((s.light_minutes || 0) / 60)}h ${(s.light_minutes || 0) % 60}m`;
        document.getElementById('txtSleepRem').textContent = `${Math.floor((s.rem_minutes || 0) / 60)}h ${(s.rem_minutes || 0) % 60}m`;
        document.getElementById('txtSleepAwake').textContent = `0h ${s.awake_minutes || 0}m`;
      } else {
        document.getElementById('valSleepTotalTime').textContent = "--";
        document.getElementById('valSleepScore').textContent = "Sin registro hoy";
        document.getElementById('barSleepDeep').style.width = "0%";
        document.getElementById('barSleepLight').style.width = "0%";
        document.getElementById('barSleepRem').style.width = "0%";
        document.getElementById('barSleepAwake').style.width = "0%";

        document.getElementById('txtSleepLight').textContent = "--";
        document.getElementById('txtSleepRem').textContent = "--";
        document.getElementById('txtSleepAwake').textContent = "--";
      }

      if (hasHr && t.hr_chart && t.hr_chart.length > 0) {
        updateChartData(t.hr_chart);
      }

      // Actualizar badge de mapeo si vino en la telemetría
      if (t.custom_steps_uuid) {
        updateMappingBadge(t.custom_steps_uuid, t.custom_steps_offset || 0, t.custom_steps_bytes || 4);

    function appendPacketToConsole(p, prepend = true) {
      if (state.isSnifferPaused) return;

      const container = document.getElementById('packetLogsStream');
      const card = document.createElement('div');
      card.className = "p-3 rounded-2xl bg-appCardElevated border border-appBorder space-y-1.5 transition-all";
      card.dataset.char = (p.short_uuid || p.char_uuid || "").toUpperCase();
      card.dataset.dir = p.direction || "RX";

      const dirColor = p.direction === "TX" ? "bg-fitTerracotta/20 text-fitTerracotta" : "bg-fitAmber/20 text-fitAmber";

      // Renderizar candidatos a pasos detectados
      let candidatesHtml = '';
      if (p.step_candidates && p.step_candidates.length > 0) {
        const items = p.step_candidates.map(c => {
          const deltaStr = c.is_increment ? `<span class="text-fitSage font-extrabold">(+${c.delta})</span>` : '';
          const targetChar = p.short_uuid || p.char_uuid || 'FEA1';
          return `
            <div class="flex items-center gap-1.5 px-2 py-0.5 rounded-lg bg-appBg border border-fitAmber/30 text-[10px]">
              <span class="text-appMuted">Off ${c.offset} [${c.type}]:</span>
              <span class="text-fitAmber font-bold">${c.value.toLocaleString()}</span>
              ${deltaStr}
              <button onclick="applyStepMapping('${targetChar}', ${c.offset}, ${c.byte_len})" class="ml-1 px-1.5 py-0.5 rounded bg-fitAmber/20 hover:bg-fitAmber text-fitAmber hover:text-black font-extrabold text-[9px] transition-all" title="Asignar este offset a pasos diarios">
                🎯 Mapear
              </button>
            </div>
          `;
        }).join('');

        candidatesHtml = `
          <div class="pt-1.5 border-t border-appBorder/50 space-y-1">
            <span class="text-[9px] text-appMuted uppercase font-bold flex items-center gap-1">
              <span>🎯</span> Candidatos detectados (Little-Endian):
            </span>
            <div class="flex flex-wrap gap-1.5">
              ${items}
            </div>
          </div>
        `;
      }

      card.innerHTML = `
        <div class="flex items-center justify-between text-[10px]">
          <div class="flex items-center gap-1.5 font-bold">
            <span class="px-1.5 py-0.5 rounded font-mono ${dirColor}">${p.direction}</span>
            <span class="text-appText">Char: ${p.short_uuid || p.char_uuid}</span>
          </div>
          <span class="text-appMuted">${p.timestamp || ""}</span>
        </div>
        <div class="text-xs font-bold text-appText tracking-wider break-all">
          <span class="text-fitSage mr-1">HEX:</span>${p.hex_string}
        </div>
        <div class="flex flex-wrap items-center justify-between text-[10px] text-appMuted pt-1 border-t border-appBorder/50">
          <span>UTF8: "${p.utf8_preview || ""}"</span>
          <span class="text-fitAmber font-bold">${p.decoded_summary || ""}</span>
        </div>
        ${p.note ? `<div class="text-[10px] text-fitMauve font-semibold">${p.note}</div>` : ''}
        ${candidatesHtml}
      `;

      if (prepend) {
        container.insertBefore(card, container.firstChild);
        if (container.children.length > 70) {
          container.removeChild(container.lastChild);
        }
      } else {
        container.appendChild(card);
      }

      applyConsoleFilter();
    }

    function applyConsoleFilter() {
      const cards = document.querySelectorAll('#packetLogsStream > div');
      cards.forEach(card => {
        const char = card.dataset.char || "";
        const dir = card.dataset.dir || "";
        if (state.packetFilter === 'ALL') {
          card.style.display = 'block';
        } else if (state.packetFilter === 'TX') {
          card.style.display = dir === 'TX' ? 'block' : 'none';
        } else {
          card.style.display = char.includes(state.packetFilter) ? 'block' : 'none';
        }
      });
    }

    async function connectDevice(address, btnElement = null) {
      const originalText = btnElement ? btnElement.innerHTML : null;
      if (btnElement) {
        btnElement.disabled = true;
        btnElement.innerHTML = `<span class="animate-spin inline-block">⏳</span> Conectando...`;
      }
      showToast(`Conectando con ${address}...`);
      try {
        const res = await fetch('/api/connect', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ address })
        });
        const data = await res.json();
        if (res.ok) {
          showToast(`¡Conectado exitosamente con ${data.name}!`);
          if (btnElement) {
            btnElement.innerHTML = `✅ Conectado`;
            btnElement.disabled = true;
          }
          fetchServices();
          fetchStatus();
          fetchTelemetry();
        } else {
          showToast(data.detail || "Error al conectar.", true);
          if (btnElement) {
            btnElement.disabled = false;
            btnElement.innerHTML = originalText;
          }
        }
      } catch (e) {
        showToast("Error de conexión BLE. Revisa permisos Bluetooth de macOS.", true);
        if (btnElement) {
          btnElement.disabled = false;
          btnElement.innerHTML = originalText;
        }
      }
    }

    function updateMappingBadge(uuid, offset, byteLen) {
      const badge = document.getElementById('currentMappingBadge');
      if (badge) {
        badge.textContent = `UUID: ${uuid.toUpperCase()} • Off: ${offset} • u${byteLen * 8}LE`;
      }
      const inpUuid = document.getElementById('mapStepsUuidInput');
      const inpOff = document.getElementById('mapStepsOffsetInput');
      const selBytes = document.getElementById('mapStepsBytesSelect');
      if (inpUuid && document.activeElement !== inpUuid) inpUuid.value = uuid.toUpperCase();
      if (inpOff && document.activeElement !== inpOff) inpOff.value = offset;
      if (selBytes && document.activeElement !== selBytes) selBytes.value = byteLen;
    }

    async function saveCustomStepsMapping() {
      const uuid = document.getElementById('mapStepsUuidInput').value.trim();
      const offset = parseInt(document.getElementById('mapStepsOffsetInput').value, 10) || 0;
      const byte_length = parseInt(document.getElementById('mapStepsBytesSelect').value, 10) || 4;
      if (!uuid) return;

      try {
        const res = await fetch('/api/characteristics/map', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ target: 'steps', uuid, offset, byte_length })
        });
        const data = await res.json();
        if (res.ok) {
          updateMappingBadge(uuid, offset, byte_length);
          showToast(`✅ Mapeo guardado: ${uuid.toUpperCase()} (Off: ${offset}, u${byte_length * 8}LE)`);
          fetchTelemetry();
        } else {
          showToast(data.detail || "Error al guardar mapeo.", true);
        }
      } catch (e) {
        showToast("Error al guardar mapeo.", true);
      }
    }

    async function applyStepMapping(uuid, offset, byteLen) {
      const inpUuid = document.getElementById('mapStepsUuidInput');
      const inpOff = document.getElementById('mapStepsOffsetInput');
      const selBytes = document.getElementById('mapStepsBytesSelect');
      if (inpUuid) inpUuid.value = uuid.toUpperCase();
      if (inpOff) inpOff.value = offset;
      if (selBytes) selBytes.value = byteLen;
      await saveCustomStepsMapping();
    }

    async function mapChar(target, uuid) {
      if (target === 'steps') {
        const inpOff = document.getElementById('mapStepsOffsetInput');
        const offset = inpOff ? parseInt(inpOff.value, 10) || 0 : 0;
        const selBytes = document.getElementById('mapStepsBytesSelect');
        const byteLen = selBytes ? parseInt(selBytes.value, 10) || 4 : 4;
        await applyStepMapping(uuid, offset, byteLen);
        return;
      }

      try {
        const res = await fetch('/api/characteristics/map', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ target, uuid })
        });
        showToast(`UUID ${uuid} asignado a ${target.toUpperCase()}`);
      } catch (e) {
        showToast("Error al mapear característica.");
      }
    }

    async function fetchDataRecords() {
      try {
        const res = await fetch('/api/data/records');
        const data = await res.json();
        const tables = data.tables || {};
        
        const hrCount = (tables.heart_rate || []).length;
        const packetCount = (tables.ble_packets || []).length;
        
        const elHr = document.getElementById('dataCountHr');
        const elPackets = document.getElementById('dataCountPackets');
        if (elHr) elHr.textContent = hrCount;
        if (elPackets) elPackets.textContent = packetCount;

        const tbody = document.getElementById('dataTableBody');
        if (!tbody) return;

        const rows = [];
        (tables.heart_rate || []).slice(0, 15).forEach(r => {
          const time = r.timestamp ? (r.timestamp.split('T')[1]?.substring(0, 8) || r.timestamp) : '--';
          rows.push(`
            <tr class="hover:bg-appCardElevated/50 transition-colors">
              <td class="py-2.5 px-2 text-appMuted">${time}</td>
              <td class="py-2.5 px-2 text-fitCoral font-bold">Frecuencia Cardíaca</td>
              <td class="py-2.5 px-2 text-appText font-bold">${r.bpm} BPM</td>
              <td class="py-2.5 px-2 text-appMuted text-[10px]">${r.source || 'BLE'}</td>
            </tr>
          `);
        });

        (tables.steps || []).slice(0, 5).forEach(s => {
          rows.push(`
            <tr class="hover:bg-appCardElevated/50 transition-colors">
              <td class="py-2.5 px-2 text-appMuted">${s.day_date || '--'}</td>
              <td class="py-2.5 px-2 text-fitAmber font-bold">Pasos del Día</td>
              <td class="py-2.5 px-2 text-appText font-bold">${s.steps} pasos (${s.calories} kcal)</td>
              <td class="py-2.5 px-2 text-appMuted text-[10px]">${s.source || 'BLE'}</td>
            </tr>
          `);
        });

        tbody.innerHTML = rows.length ? rows.join('') : `<tr><td colspan="4" class="py-4 text-center text-appMuted">Sin registros en la base de datos aún.</td></tr>`;
      } catch (e) {
        console.error("Error al obtener registros SQLite:", e);
      }
    }

#!/usr/bin/env bash
set -e

DIR="$( cd "$( dirname "${BASH_SOURCE[0]}" )" >/dev/null 2>&1 && pwd )"
cd "$DIR"

echo "=========================================================="
echo "          🚀 Iniciando Fit3 Mac Studio                    "
echo "    Samsung Galaxy Fit 3 (SM-R390) • BLE macOS Monterey   "
echo "=========================================================="

# 1. Comprobar Python 3
if ! command -v python3 &> /dev/null; then
    echo "❌ Error: Python 3 no está instalado en este Mac."
    exit 1
fi

# 2. Crear entorno virtual si no existe
if [ ! -d "venv" ]; then
    echo "📦 Creando entorno virtual 'venv' en Python 3..."
    python3 -m venv venv
fi

# 3. Activar venv e instalar dependencias
echo "🔄 Verificando librerías (FastAPI, Bleak, Uvicorn)..."
venv/bin/pip install -r requirements.txt --quiet

echo ""
echo "✅ Servidor preparado para recolectar y visualizar métricas:"
echo "   💻 Panel en tu Mac:  http://localhost:8000"
echo "=========================================================="
echo ""

# 5. Abrir la app en el navegador del Mac en segundo plano
(sleep 1.5 && open "http://localhost:8000" 2>/dev/null || true) &

# 6. Lanzar servidor FastAPI escuchando en 0.0.0.0:8000
exec venv/bin/uvicorn main:app --host 0.0.0.0 --port 8000 --reload
